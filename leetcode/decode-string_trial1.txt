class Solution:
    def decodeString(self, s: str) -> str:
        st = []
        
        for c in s:
            if c != ']':
                st.append(c)
            else:
                temp = ""
                while st[-1] != '[':
                    temp = st.pop() + temp
                
                st.pop()  
                num = ""
                while st and st[-1].isdigit():
                    num = st.pop() + num
                st.append(int(num) * temp)
        
        return "".join(st)
