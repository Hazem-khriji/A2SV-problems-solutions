class Solution:
    def validPalindrome(self, s: str) -> bool:
        n = len(s)
        l,r  = 0 ,n-1
        used = False
        def check_palind(s,l,r):
            while(l<r):
                if(s[l]!=s[r]):
                    return False
                l+=1
                r-=1
            return True
        while(l<r):
            if(s[l]==s[r]):
                l+=1
                r-=1
            else:
                if(used):
                    return False
                used = True
                return check_palind(s,l+1,r) or check_palind(s,l,r-1)
        return True
