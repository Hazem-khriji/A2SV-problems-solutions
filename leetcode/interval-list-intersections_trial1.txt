class Solution:
    def intervalIntersection(self, firstList: List[List[int]], secondList: List[List[int]]) -> List[List[int]]:
        f,s=0,0
        nf,ns=len(firstList), len(secondList)
        res = []
        while(f<nf and s<ns):
            first1=firstList[f][0]
            second1=firstList[f][-1]
            first2=secondList[s][0]
            second2=secondList[s][-1]
            if(first1>second2):
                s+=1
            elif(first2>second1):
                f+=1
            else:
                intersect=[max(first1,first2),min(second1,second2)]
                res.append(intersect)
                if(second1>second2):
                    s+=1
                elif(second1<second2):
                    f+=1
                else:
                    f+=1
                    s+=1
        return res
