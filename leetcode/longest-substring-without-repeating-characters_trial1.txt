class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        l=0
        n=len(s)
        ss=set()
        maxi=0
        for r in range(len(s)):
            c=s[r]
            if(s[r] not in ss):
                ss.add(c)
            else:
                maxi =max(maxi,r-l)
                while(s[l]!=c):
                    ss.discard(s[l])
                    l+=1
                l+=1
        maxi=max(maxi,n-l)
        return maxi
