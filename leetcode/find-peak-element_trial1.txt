class Solution:
    def findPeakElement(self, nums: List[int]) -> int:
        n = len(nums)
        low , high = 0 , n - 1
        while(low<=high):
            mid = low + (high-low)//2
            num = nums[mid]
            right = nums[mid+1] if mid < n-1 else nums[mid]
            left = nums[mid-1] if mid > 0 else nums[mid]
            if num>= right and num>=left:
                return mid
            elif num < right:
                low = mid + 1
            else:
                high = mid - 1 
        return low
