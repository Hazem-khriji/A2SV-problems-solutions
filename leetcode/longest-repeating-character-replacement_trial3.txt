class Solution:
    def characterReplacement(self, s: str, k: int) -> int:
        hash_map = {}
        max_freq = 0
        longest = 0
        left = 0
        n = len(s)

        for right in range(n):
            ch = s[right]

            hash_map[ch] = hash_map.get(ch,0) + 1
            max_freq = max(max_freq,hash_map[ch])

            if (right - left + 1) - max_freq > k:
                hash_map[s[left]] -= 1
                left += 1
            longest = max(longest, right - left + 1)
        
        return longest
