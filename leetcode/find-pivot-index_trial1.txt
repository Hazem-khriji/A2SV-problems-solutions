class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        pre=[nums[0]]
        suff=[0]*len(nums)
        suff[-1]=nums[-1]
        for i in range(1,len(nums)):
            j=-i-1
            pre.append(pre[i-1]+nums[i])
            suff[j]=suff[j+1]+nums[j]
        if(len(nums)==1 or suff[1]==0):
            return 0
        for i in range(1,len(pre)-1):
            if(pre[i-1]==suff[i+1]):
                return i
        if(pre[-2]==0):
            return len(pre)-1
        return -1
