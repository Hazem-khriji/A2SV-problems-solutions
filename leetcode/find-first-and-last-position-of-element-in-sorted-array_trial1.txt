class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        n = len(nums)
        l , r = 0 , n - 1
        mid = -1
        pos = -1
        while(l<=r):
            mid = (l + r) // 2
            if(nums[mid] == target):
                pos = mid
                r = mid - 1
            elif nums[mid] > target:
                r = mid - 1
            else:
                l = mid + 1
        res = [pos]
        l , r = 0 , n - 1
        pos = -1
        while(l<=r):
            mid = (l + r) // 2
            if(nums[mid] == target):
                pos = mid
                l = mid + 1
            elif nums[mid] < target:
                l = mid  + 1
            else:
                r = mid - 1
        res.append(pos)
        return res
