class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        n = len(s)
        m = len(p)
        dp = {}
        def f(i, j):
            if (i, j) in dp:
                return dp[(i, j)]

            if j == m:
                return i == n

            match = i < n and (s[i] == p[j] or p[j] == '.')

            if j+1 < m and p[j+1] == '*':
                ans = f(i, j+2) or (match and f(i+1, j))
            else:
                ans = match and f(i+1, j+1)

            dp[(i,j)] = ans
            return ans
        return f(0,0)
