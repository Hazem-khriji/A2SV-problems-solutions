class Solution:
    def threeSum(self, nums: list[int]) -> list[list[int]]:
        nums.sort()
        res=[]
        for i in range(len(nums)-1):
            if(i>=1 and nums[i]==nums[i-1]):
                continue
            t = -nums[i]
            j,k=i+1,len(nums)-1
            while(j<k):
                if(nums[j]+nums[k]<t):
                    j+=1
                elif(nums[j]+nums[k]>t):
                    k-=1
                else:
                    sol=[nums[i],nums[j],nums[k]]
                    res.append(sol)
                    j+=1
                    k-=1
                    while(j<k and nums[j]==nums[j-1] and nums[k]==nums[k+1]):
                        j+=1
                        k-=1
                    

        return res
