from bisect import bisect_left, bisect_right
class MyCalendar:

    def __init__(self):
        self.starts =  []
        self.ends = []
        self.length = 0
        

    def book(self, startTime: int, endTime: int) -> bool:
        if(self.length == 0):
            self.starts.append(startTime)
            self.ends.append(endTime)
            self.length += 1
            return True

        index = bisect_right(self.ends,startTime)
        if index >= self.length :
            self.starts.append(startTime)
            self.ends.append(endTime)
            self.length += 1
            return True
        starting = self.starts[index]
        if starting<endTime : 
            return False
        else:
            self.starts = self.starts[:index]+[startTime]+self.starts[index::]
            self.ends = self.ends[:index]+[endTime]+self.ends[index::]
            self.length += 1
            return True
