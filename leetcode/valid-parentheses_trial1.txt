class Solution:
    def isValid(self, s: str) -> bool:
        from collections import deque

        stack = deque()
        for i in range (len(s)):
            if(s[i]=='(' or s[i]=='[' or s[i]=='{'):
                stack.append(s[i])
            else:
                if(len(stack)==0):
                    return False
                else:
                    k=stack.pop()
                    if((s[i]==')' and not(k=='(')) or (s[i]==']' and not(k=='[')) or (s[i]=='}' and not(k=='{'))):
                        return False
            
        return len(stack)==0
