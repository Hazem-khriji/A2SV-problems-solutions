class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        n = len(matrix)
        m = len(matrix[0])
        lr , rr = 0 , n - 1
        if (target<matrix[0][0] or target>matrix[n-1][m-1]):
            return False
        while(lr <= rr):
            mid = (lr + rr) // 2
            start = matrix[mid][0]
            end = matrix[mid][-1]
            if(start<=target<=end):
                lr = mid
                break
            elif(start>target):
                rr = mid - 1 
            else:
                lr = mid + 1
        index = bisect_left(matrix[lr],target)
        if(index>=m or matrix[lr][index] != target):
            return False
        else:
            return True
