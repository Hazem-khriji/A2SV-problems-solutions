from collections import deque
class RecentCounter:

    def __init__(self):
        self.counter = deque()
        self.first = 0

    def ping(self, t: int) -> int :
        self.counter.append(t)
        while(t-self.counter[self.first] > 3000):
            self.first += 1
        return (len(self.counter)-self.first)
