from collections import Counter, defaultdict

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        if len(s) < len(t):
            return ""
        c = Counter(t)
        d = defaultdict(int)
        have = 0
        need = len(c)
        res = ""
        l = 0
        for r in range(len(s)):
            char = s[r]
            d[char] += 1
            if char in c and d[char] == c[char]:
                have += 1

            while have == need:
                if res == "" or (r - l + 1) < len(res):
                    res = s[l:r+1]
                d[s[l]] -= 1
                if s[l] in c and d[s[l]] < c[s[l]]:
                    have -= 1
                l += 1
        return res
