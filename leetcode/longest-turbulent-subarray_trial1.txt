from typing import List

class Solution:
    def maxTurbulenceSize(self, arr: List[int]) -> int:
        n = len(arr)
        if n < 2:
            return n

        l = 0
        res = 1

        for r in range(1, n):
            if arr[r] == arr[r-1]:
                l = r
            elif r == 1 or (arr[r] - arr[r-1]) * (arr[r-1] - arr[r-2]) >= 0:
                l = r - 1

            res = max(res, r - l + 1)

        return res
