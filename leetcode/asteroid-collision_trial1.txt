class Solution:
    def asteroidCollision(self, asteroids: List[int]) -> List[int]:
        res=[]
        flag = 1
        for a in asteroids:
            flag=1
            if a>0:
                res.append(a)
            else:
                while(len(res)>0 and res[-1]>0):
                    if(abs(a)>res[-1]):
                        res.pop()
                    elif(abs(a)==res[-1]):
                        res.pop()
                        flag = 0
                        break
                    else:
                        flag=0
                        break
                if(flag==1 and (len(res)==0 or res[-1]<0)):
                    res.append(a)
        return res
