class Solution:
    def maxDistance(self, position: List[int], m: int) -> int:
        position.sort()
        n = len(position)
        def works(x):
            k = m - 1
            prev = position[0]
            for pos in position:
                if (pos - prev >= x):
                    prev = pos
                    k -= 1
            return k<=0
        l , r = 0 , max(position)
        while(l<=r):
            mid = l + (r - l) // 2
            if works(mid):
                l = mid  + 1
            else:
                r = mid - 1
        return l-1
