class Solution:
    def predictTheWinner(self, nums: List[int]) -> bool:
        n = len(nums)
        def f(l,r):
            if(l>=r):
                return  nums[r]
            return max(
                nums[l] - f(l+1, r),
                nums[r] - f(l, r-1)
            )
            
            
        return f(0,n-1)>=0
