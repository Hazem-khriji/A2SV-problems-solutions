class StockSpanner:

    def __init__(self):
        self.span=[]

    def next(self, price: int) -> int:
        res = 0
        i = len(self.span) - 1
        while(i>=0 and self.span[i][0]<=price):
            i-=1
            k = self.span.pop()
            res+=k[1]
        self.span.append((price,res+1))
        return res+1
