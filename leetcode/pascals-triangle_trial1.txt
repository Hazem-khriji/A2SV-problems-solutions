class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 1:
            return [[1]]
        if numRows == 2:
            return [[1],[1,1]]        
        res = [[1],[1,1]]
        for _ in range(numRows-2):
            prev = res[-1]
            n = len(prev)+1
            curr = [1]*n
            for i in range(1,n//2):
                j = n - 1 - i
                s = prev[i]+prev[i-1]
                curr[i],curr[j] = s,s
            if(n%2 != 0):
                curr[n//2] = 2 * prev[n//2-1]
            res.append(curr)
        return res
