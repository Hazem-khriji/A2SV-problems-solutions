class Solution:
    def numRescueBoats(self, people: List[int], limit: int) -> int:
        n = len(people)
        if(n==1):
            return 1
        people.sort()
        res = 0
        l , r = 0 , n-1
        while l<=r:
            if(people[l]+people[r]<=limit):
                l+=1
            r-=1
            res+=1
        return res
