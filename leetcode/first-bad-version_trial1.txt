# def isBadVersion(version: int) -> bool:

class Solution:
    def firstBadVersion(self, n: int) -> int:
        l , r = 1 , n
        res = 1
        while(l < r):
            mid = (l + r) // 2
            if(isBadVersion(mid)):
                r = mid 
                res = mid
            else:
                l = mid + 1
                res = mid + 1
        return res
