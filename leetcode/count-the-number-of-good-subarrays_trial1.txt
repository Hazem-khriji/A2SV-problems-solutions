from collections import defaultdict

class Solution:
    def countGood(self, nums, k):
        d = defaultdict(int)
        n = len(nums)
        l = 0
        res = 0
        count = 0

        for r in range(n):
            num = nums[r]

            count += d[num]
            d[num] += 1

            while count >= k:
                x = nums[l]
                d[x] -= 1
                count -= d[x]
                l += 1

            res += l

        return res
