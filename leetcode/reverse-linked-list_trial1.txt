# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if(head == None or head.next == None):
            return head
        reverse = None
        while(head!=None):
            temp = ListNode(head.val,reverse)
            reverse = temp
            head = head.next
        return reverse
