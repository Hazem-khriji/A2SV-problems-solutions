class Solution:
    def totalFruit(self, fruits: List[int]) -> int:
        last_seen={}
        maxi=0
        l=0
        for r in range(len(fruits)):
            if(fruits[r] not in last_seen):
                if(len(last_seen)==2):
                    deleted=min(last_seen.values())
                    l = deleted + 1
                    for k,v in last_seen.items():
                        if v==deleted:
                            last_seen.pop(k)
                            break    
            last_seen[fruits[r]]=r
            maxi = max(maxi,r-l+1)
        return maxi
