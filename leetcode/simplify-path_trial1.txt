class Solution:
    def simplifyPath(self, path: str) -> str:
        st = []
        l = path.split('/')
        for part in l:
            if(not part or part == '.' or part == '/'):
                continue
            elif(part == ".."):
                if(len(st)>=1):
                    st.pop()
            else:
                st.append(part)
        return '/'+'/'.join(st)
