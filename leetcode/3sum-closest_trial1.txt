class Solution:
    def threeSumClosest(self, nums: List[int], target: int) -> int:
        nums.sort()
        n=len(nums)
        mini=nums[0]+nums[1]+nums[2]
        for i in range(len(nums)):
            j,k=i+1,n-1
            while j<k:
                s=nums[i]+nums[j]+nums[k]
                if(abs(mini-target)>abs(s-target)):
                    mini=s
                if s>target:
                    k=k-1
                elif s<target:
                    j+=1
                else:
                    return s
                
        return mini
