class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        c=Counter(s1)
        n=len(s2)
        k = n - len(s1)
        if(k<0):
            return False
        d = Counter(s2[:len(s1)])
        if(c==d):
            return True
        l,r = 1,len(s1)
        while(r<n):
            char_add=s2[r]
            char_rem=s2[l-1]
            d[char_add] = d.get(char_add, 0) + 1
            d[char_rem] -= 1
            if(d[char_rem]==0):
                del d[char_rem]
            if(d==c):
                return True
            l+=1
            r+=1
        return False
