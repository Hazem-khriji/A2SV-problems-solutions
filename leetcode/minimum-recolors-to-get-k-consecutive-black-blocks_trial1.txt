class Solution:
    def minimumRecolors(self, blocks: str, k: int) -> int:
        n=len(blocks)
        res=n
        count=0
        l=0
        for r in range(n):
            if(blocks[r]=="W"):
                count+=1
            if(r>=k-1):
                res=min(res,count)
                if(blocks[l]=='W'):
                    count-=1
                l+=1
        return res
