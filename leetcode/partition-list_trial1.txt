# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def partition(self, head: Optional[ListNode], x: int) -> Optional[ListNode]:
        list_less = ListNode()
        start_less = list_less
        list_more = ListNode()
        start_more = list_more
        while(head):
            d = ListNode(head.val)
            if head.val<x:
                list_less.next = d
                list_less = list_less.next
            else:
                list_more.next = d
                list_more = list_more.next
            head = head.next
        list_less.next = start_more.next
        return start_less.next
