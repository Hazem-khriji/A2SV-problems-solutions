"""
# Definition for a Node.
class Node:
    def __init__(self, val, prev, next, child):
        self.val = val
        self.prev = prev
        self.next = next
        self.child = child
"""

class Solution:
    def flatten(self, head: 'Optional[Node]') -> 'Optional[Node]':

        def dfs(node):
            curr = node
            last = node

            while curr:
                nxt = curr.next

                if curr.child:
                    tail = dfs(curr.child)
                    curr.next = curr.child
                    curr.child.prev = curr
                    if nxt:
                        tail.next = nxt
                        nxt.prev = tail

                    curr.child = None
                    last = tail
                else:
                    last = curr
                curr = nxt
            return last
        if head:
            dfs(head)
        return head
