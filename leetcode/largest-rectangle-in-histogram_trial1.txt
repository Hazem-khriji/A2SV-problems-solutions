class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        s =  []
        n = len(heights)
        maxi = 0
        for i in range(n):
            h = heights[i]
            beg = i
            while(s and s[-1][1]> h):
                k = s.pop()
                maxi = max(maxi,(i-k[0])*k[1])
                beg = k[0]
            s.append((beg,h))
        while(s):
            k = s.pop()
            maxi = max(maxi,(n-k[0])*k[1])
        return maxi
