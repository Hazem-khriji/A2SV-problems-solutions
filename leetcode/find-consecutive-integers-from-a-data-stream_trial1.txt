        self.value = value
        self.k = k
        self.cons = 0

    def consec(self, num: int) -> bool:
        self.stream.append(num)
        if(num!= self.value):
            self.cons = 0
            return False
        else:
            self.cons += 1
            if self.cons >= self.k:
                return True
            else:
                return False
